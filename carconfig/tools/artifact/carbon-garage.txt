const nameWords = (s) =>
  String(s || "").replace(/([a-z])([A-Z])/g, "$1 $2").replace(/([A-Za-z])(\d)/g, "$1 $2")
    .replace(/(\d)([A-Za-z])/g, "$1 $2").replace(/[_.\-:]+/g, " ").toLowerCase();

const LAMP_WORDS = "head ?lights?|tail ?lights?|lamps?|lights?|indicators?|blinkers?|phares?|feux|faros?|fanali|luce|luz|scheinwerfer|leuchten?";
const KINDS = [
  ["lens", new RegExp(`\\b(${LAMP_WORDS}) ?(glass|lens|cover|vitre|verre|vetro|vidrio|glas)|\\blens(es)?\\b`)],
  ["glass", /\b(glass|windows?|windscreen|windshield|backlight|vitres?|verre|vetro|vetri\w*|vidrios?|cristal(es)?|glas|scheiben?|ventanas?|finestrini)\b/],
  ["tyre", /\b(tyres?|tires?|rubber|tread|sidewall|pneus?|pneumatici|gomma|gomme|goma|neumaticos?|reifen)\b/],
  ["carbon", /\b(carbon|cfrp|carbone|carbono)\b/],
  ["chrome", /\b(chrome|chromed|polished|chromo|cromo|cromato|chrom)\b/],
  ["plastic", /\b(plastic|trim ?black|black ?trim|textured|matte ?black|grille|grill|plastique|plastica|plastico|kunststoff)\b/],
];
const kindOf = (name) => { const w = nameWords(name); for (const [k, re] of KINDS) if (re.test(w)) return k; return null; };

function normalise(m) {
  const kind = kindOf(m.name);
  if (!kind || !m.isMeshStandardMaterial) return m;
  const p = m.clone();
  if (kind === "glass") Object.assign(p, { metalness: 0, roughness: 0.02, transparent: true, opacity: 0.38, depthWrite: false, side: THREE.DoubleSide }), p.color.set("#1c2528");
  if (kind === "lens") Object.assign(p, { metalness: 0, roughness: 0.03, transparent: true, opacity: 0.22, depthWrite: false }), p.color.set("#ffffff");
  if (kind === "tyre") { p.metalness = 0; p.roughness = 0.9; p.color.lerp(new THREE.Color("#141414"), 0.6); }
  if (kind === "chrome") { p.metalness = 1; p.roughness = 0.05; p.color.set("#e4e7ea"); }
  if (kind === "carbon") { p.metalness = 0.1; p.roughness = 0.35; }
  if (kind === "plastic") { p.metalness = 0; p.roughness = Math.max(p.roughness, 0.6); }
  return p;
}

const paintNameSays = (name) => { const w = nameWords(name); if (OTHER_WORDS.test(w)) return "other"; if (PAINT_WORDS.test(w)) return "paint"; return null; };

function choosePaint(stats) {
  const total = stats.reduce((s, m) => s + m.area, 0);
  if (total <= 0) return { how: "none", ids: [] };
  const share = (m) => m.area / total;
  const lum = ([r, g, b]) => 0.2126 * r + 0.7152 * g + 0.0722 * b;
  const named = stats.filter((m) => m.kind === null && paintNameSays(m.name) === "paint" && share(m) > 0.02 && (lum(m.rgb) > 0.04 || share(m) > 0.08));
  if (named.length) return { how: "named", ids: named.map((m) => m.id) };
  const surfaces = stats.filter((m) => m.kind === null && paintNameSays(m.name) !== "other" && m.opacity >= 0.95 && m.height >= 0.25 && m.height <= 0.8);
  const panels = surfaces.filter((m) => m.span >= 0.55);
  if (!panels.length) return { how: "none", ids: [] };
  const best = panels.reduce((a, b) => (b.area > a.area ? b : a));
  if (share(best) < 0.12 || share(best) > 0.85) return { how: "none", ids: [] };
  if (best.hasMap && /\b(atlas|baked?)\b/.test(nameWords(best.name))) return { how: "none", ids: [] };
  const dist = (a, b) => Math.hypot(a[0] - b[0], a[1] - b[1], a[2] - b[2]);
  const isDefault = (c) => [1, 0.8].some((v) => c.every((x) => Math.abs(x - v) < 1e-3));
  const siblings = best.hasMap || isDefault(best.rgb) ? [] : surfaces.filter((m) => m !== best && !m.hasMap && m.span >= 0.15 && dist(m.rgb, best.rgb) < 0.03);
  return { how: "largest", ids: [best.id, ...siblings.map((m) => m.id)] };
}

const materialsOf = (mesh) => (Array.isArray(mesh.material) ? mesh.material : [mesh.material]);

function dropFloor(model) {
  model.updateMatrixWorld(true);
  const meshes = [];
  model.traverse((o) => { if (o.isMesh) meshes.push({ o, box: new THREE.Box3().setFromObject(o) }); });
  const floors = new Set();
  for (const { o, box } of meshes) {
    const rest = new THREE.Box3();
    for (const m of meshes) if (m.o !== o && !floors.has(m.o)) rest.union(m.box);
    if (rest.isEmpty()) continue;
    const r = rest.getSize(new THREE.Vector3()), s = box.getSize(new THREE.Vector3());
    if (s.x > r.x * 1.05 && s.z > r.z * 1.05 && box.max.y < rest.min.y + r.y * 0.2) floors.add(o);
  }
  for (const o of floors) o.removeFromParent();
}

function materialStats(root) {
  root.updateMatrixWorld(true);
  const box = new THREE.Box3().setFromObject(root), size = box.getSize(new THREE.Vector3());
  const acc = new Map();
  let triangles = 0;
  root.traverse((o) => { if (o.isMesh) triangles += (o.geometry.index?.count ?? o.geometry.attributes.position?.count ?? 0) / 3; });
  const step = Math.max(1, Math.floor(triangles / 150000));
  const a = new THREE.Vector3(), b = new THREE.Vector3(), c = new THREE.Vector3(), ab = new THREE.Vector3(), ac = new THREE.Vector3();
  root.traverse((o) => {
    if (!o.isMesh || !o.visible) return;
    const g = o.geometry, pos = g.attributes.position; if (!pos) return;
    const index = g.index, count = index ? index.count : pos.count, mats = materialsOf(o);
    const groups = Array.isArray(o.material) && g.groups.length ? g.groups : [{ start: 0, count, materialIndex: 0 }];
    for (const grp of groups) {
      const m = mats[grp.materialIndex ?? 0]; if (!m) continue;
      let e = acc.get(m); if (!e) acc.set(m, (e = { area: 0, h: 0, lo: Infinity, hi: -Infinity }));
      const end = Math.min(grp.start + grp.count, count);
      for (let i = grp.start; i + 2 < end; i += 3 * step) {
        a.fromBufferAttribute(pos, index ? index.getX(i) : i).applyMatrix4(o.matrixWorld);
        b.fromBufferAttribute(pos, index ? index.getX(i + 1) : i + 1).applyMatrix4(o.matrixWorld);
        c.fromBufferAttribute(pos, index ? index.getX(i + 2) : i + 2).applyMatrix4(o.matrixWorld);
        const area = 0.5 * ab.subVectors(b, a).cross(ac.subVectors(c, a)).length() * step;
        e.area += area; e.h += (area * ((a.y + b.y + c.y) / 3 - box.min.y)) / (size.y || 1);
        e.lo = Math.min(e.lo, a.z, b.z, c.z); e.hi = Math.max(e.hi, a.z, b.z, c.z);
      }
    }
  });
  const materials = [...acc.keys()];
  const stats = materials.map((m, id) => {
    const e = acc.get(m);
    return { id, name: m.name, kind: kindOf(m.name), rgb: m.color ? [m.color.r, m.color.g, m.color.b] : [1, 1, 1], opacity: m.transparent ? m.opacity : 1,
      hasMap: Boolean(m.map), area: e.area, span: e.area > 0 ? (e.hi - e.lo) / (size.z || 1) : 0, height: e.area > 0 ? e.h / e.area : 0 };
  });
  return { materials, stats };
}

const cornerKey = (c) => `${c.z > 0 ? "f" : "r"}${c.x > 0 ? "r" : "l"}`;
function sane(groups) {
  const found = ["fl", "fr", "rl", "rr"].map((k) => groups.get(k));
  const ok = found.every((g) => { if (!g) return false; const s = g.box.getSize(new THREE.Vector3()); const r = s.y / 2; return r > 0.2 && r < 0.52 && Math.abs(s.z - s.y) / s.y < 0.3; });
  return ok ? found : null;
}
const WHEEL_RE = /\b(wheel|rim|tyre|tire|brake|caliper|disc|rotor|lug|hub|spoke|rueda|roue|ruota|llanta|jante|felge|reifen|gomma|pneu)/i;

function splitAcrossCorners(mesh) {
  if (Array.isArray(mesh.material) || !mesh.parent) return;
  const g = mesh.geometry.index ? mesh.geometry.toNonIndexed() : mesh.geometry;
  const pos = g.attributes.position; if (!pos) return;
  const tris = Math.floor(pos.count / 3), corner = new Uint8Array(tris), counts = [0, 0, 0, 0];
  const v = new THREE.Vector3(), c = new THREE.Vector3();
  mesh.updateMatrixWorld(true);
  for (let t = 0; t < tris; t++) {
    c.set(0, 0, 0);
    for (let k = 0; k < 3; k++) c.add(v.fromBufferAttribute(pos, t * 3 + k).applyMatrix4(mesh.matrixWorld));
    const key = (c.x > 0 ? 1 : 0) | (c.z > 0 ? 2 : 0);
    corner[t] = key; counts[key]++;
  }
  if (counts.some((n) => n === 0)) return;
  for (let key = 0; key < 4; key++) {
    const geo = new THREE.BufferGeometry();
    for (const [name, attr] of Object.entries(g.attributes)) {
      const inter = attr.isInterleavedBufferAttribute, src = inter ? attr.data.array : attr.array;
      const stride = inter ? attr.data.stride : attr.itemSize, offset = inter ? attr.offset : 0, size = attr.itemSize;
      const out = new src.constructor(counts[key] * 3 * size);
      let w = 0;
      for (let t = 0; t < tris; t++) {
        if (corner[t] !== key) continue;
        for (let k = 0; k < 3; k++) { const base = (t * 3 + k) * stride + offset; for (let i = 0; i < size; i++) out[w++] = src[base + i]; }
      }
      geo.setAttribute(name, new THREE.BufferAttribute(out, size, attr.normalized));
    }
    geo.computeBoundingBox(); geo.computeBoundingSphere();
    const piece = new THREE.Mesh(geo, mesh.material);
    piece.name = `${mesh.name} ${key}`;
    piece.position.copy(mesh.position); piece.quaternion.copy(mesh.quaternion); piece.scale.copy(mesh.scale);
    piece.castShadow = piece.receiveShadow = true;
    mesh.parent.add(piece);
  }
  mesh.removeFromParent();
}

function label(o) {
  const names = [];
  for (let p = o; p; p = p.parent) if (p.name) names.push(nameWords(p.name));
  return `${names.join(" ")} ${materialsOf(o).map((m) => nameWords(m.name)).join(" ")}`;
}

function wheelsByName(root, carLength) {
  const merged = [];
  root.traverse((o) => {
    if (!o.isMesh) return;
    const l = label(o);
    if (!WHEEL_RE.test(l) || /steering|spare|light|lamp|arch|well/i.test(l)) return;
    const b = new THREE.Box3().setFromObject(o), bs = b.getSize(new THREE.Vector3()), bc = b.getCenter(new THREE.Vector3());
    if (bs.y <= 0.95 && bc.y <= 0.6 && bs.x > 0.6 && bs.z > carLength * 0.4) merged.push(o);
  });
  merged.forEach(splitAcrossCorners);
  root.updateMatrixWorld(true);
  const groups = new Map();
  let tangled = false;
  root.traverse((o) => {
    if (!o.isMesh) return;
    const l = label(o);
    if (!WHEEL_RE.test(l) || /steering|spare|light|lamp|arch|well/i.test(l)) return;
    const b = new THREE.Box3().setFromObject(o), bs = b.getSize(new THREE.Vector3()), bc = b.getCenter(new THREE.Vector3());
    if (bs.y > 0.95 || bc.y > 0.6) return;
    if ((b.min.x < 0 && b.max.x > 0 && bs.x > 0.6) || (b.min.z < 0 && b.max.z > 0 && bs.z > carLength * 0.4)) tangled = true;
    if (Math.abs(bc.x) < 0.15) return;
    const key = cornerKey(bc), g = groups.get(key);
    if (g) { g.box.union(b); g.meshes.push(o); } else groups.set(key, { box: b, meshes: [o] });
  });
  return tangled ? null : sane(groups);
}

function wheelsByShape(root) {
  root.updateMatrixWorld(true);
  const whole = new THREE.Box3().setFromObject(root), width = whole.getSize(new THREE.Vector3()).x;
  const meshes = [];
  root.traverse((o) => { if (o.isMesh) meshes.push({ o, box: new THREE.Box3().setFromObject(o) }); });
  const tyres = new Map();
  for (const m of meshes) {
    const s = m.box.getSize(new THREE.Vector3()), c = m.box.getCenter(new THREE.Vector3());
    if (!(Math.abs(s.z - s.y) / s.y < 0.2 && s.y > 0.45 && s.y < 1 && s.x < s.y * 0.6 && m.box.min.y - whole.min.y < 0.06 && Math.abs(c.x) > width * 0.2)) continue;
    const key = cornerKey(c), prev = tyres.get(key);
    if (!prev || s.y > prev.box.getSize(new THREE.Vector3()).y) tyres.set(key, m);
  }
  if (tyres.size !== 4) return null;
  const groups = new Map();
  for (const [key, tyre] of tyres) {
    const zone = tyre.box.clone(); zone.min.x -= 0.12; zone.max.x += 0.12; zone.expandByScalar(0.01);
    const parts = meshes.filter((m) => m === tyre || zone.containsBox(m.box)).map((m) => m.o);
    const box = new THREE.Box3(); parts.forEach((p) => box.expandByObject(p));
    groups.set(key, { box, meshes: parts });
  }
  return sane(groups);
}

const FRONT = [[/\b(head ?lights?|head ?lamps?|headlamps?)\b/, 3], [/\bsteering\b/, 3], [/\b(grille?|grill|radiator)\b/, 2], [/\b(windshield|windscreen|wipers?|dashboard|dash)\b/, 2], [/\b(bonnet|hood|splitter)\b/, 1], [/\bfront\b/, 1]];
const REAR = [[/\b(tail ?lights?|tail ?lamps?|taillamps?|brake ?lights?|reverse ?lights?)\b/, 3], [/\b(exhausts?|muffler|tailpipes?)\b/, 2], [/\b(trunk|boot|diffuser)\b/, 2], [/\b(rear|back)\b/, 1]];
const LAMP_RE = /\b(head ?lights?|head ?lamps?|headlamps?|tail ?lights?|tail ?lamps?|brake ?lights?|drl|leds?|lights?|lamps?|phares?|faros?|feux|fanali|scheinwerfer|licht|leuchten?)\b/;

function prepare(gltf, entry) {
  const model = gltf.scene;
  const copies = new Map();
  const own = (m) => { let c = copies.get(m); if (!c) copies.set(m, (c = normalise(m))); return c; };
  model.traverse((o) => { if (!o.isMesh) return; o.material = Array.isArray(o.material) ? o.material.map(own) : own(o.material); o.castShadow = o.receiveShadow = true; });
  dropFloor(model);

  const orient = new THREE.Group(); orient.add(model);
  const root = new THREE.Group(); root.add(orient);
  let box = new THREE.Box3().setFromObject(root), size = box.getSize(new THREE.Vector3());
  orient.rotation.y = size.x > size.z ? Math.PI / 2 : 0;
  root.updateMatrixWorld(true);

  // Front toward +z, from what the parts are called.
  box = new THREE.Box3().setFromObject(root);
  const mid = box.getCenter(new THREE.Vector3()).z, len = box.getSize(new THREE.Vector3()).z;
  let vote = 0;
  root.traverse((o) => {
    if (!o.isMesh) return;
    const along = (new THREE.Box3().setFromObject(o).getCenter(new THREE.Vector3()).z - mid) / len;
    if (Math.abs(along) < 0.08) return;
    const l = label(o), side = Math.sign(along);
    for (const [re, w] of FRONT) if (re.test(l)) vote += w * side;
    for (const [re, w] of REAR) if (re.test(l)) vote -= w * side;
  });
  if (vote < 0) { orient.rotation.y += Math.PI; root.updateMatrixWorld(true); }

  box = new THREE.Box3().setFromObject(root); size = box.getSize(new THREE.Vector3());
  const target = entry.length ?? (size.z > 3 && size.z < 6.5 ? size.z : 4.6);
  orient.scale.multiplyScalar(target / size.z);
  root.updateMatrixWorld(true);
  box = new THREE.Box3().setFromObject(root);
  const centre = box.getCenter(new THREE.Vector3());
  orient.position.set(-centre.x, -box.min.y, -centre.z);
  root.updateMatrixWorld(true);

  const wheels = new THREE.Group();
  const found = wheelsByName(root, target) ?? wheelsByShape(root);
  if (found) for (const g of found) for (const m of g.meshes) wheels.attach(m);

  const slotsWhere = (test) => {
    const slots = [];
    root.traverse((o) => { if (o.isMesh) materialsOf(o).forEach((m, i) => { if (test(m, o)) slots.push({ mesh: o, index: Array.isArray(o.material) ? i : null, original: m }); }); });
    return slots;
  };
  const { materials, stats } = materialStats(root);
  const choice = choosePaint(stats);
  const chosen = new Set(choice.ids.map((i) => materials[i]));
  const paintSlots = slotsWhere((m) => chosen.has(m));
  const glassSlots = slotsWhere((m) => !chosen.has(m) && kindOf(m.name) === "glass");
  const lampSlots = slotsWhere((m, mesh) => !chosen.has(m) && (LAMP_RE.test(nameWords(m.name)) || LAMP_RE.test(label(mesh))))
    .map((s) => ({ ...s, front: new THREE.Box3().setFromObject(s.mesh).getCenter(new THREE.Vector3()).z >= 0 }));

  const body = new THREE.Group(); body.add(root);
  const car = new THREE.Group(); car.add(body, wheels);
  const finalBox = new THREE.Box3().setFromObject(car);
  return { car, body, hasWheels: Boolean(found), paintSlots: paintSlots.length ? paintSlots : [], paintable: paintSlots.length > 0, glassSlots, lampSlots, size: finalBox.getSize(new THREE.Vector3()) };
}

function setSlots(slots, material) {
  for (const s of slots) {
    const m = material ?? s.original;
    if (s.index === null) s.mesh.material = m; else s.mesh.material[s.index] = m;
  }
}

function paintMaterial(hex, finish) {
  const base = { color: new THREE.Color(hex), envMapIntensity: 1.25 };
  const P = (o) => new THREE.MeshPhysicalMaterial({ ...base, ...o });
  switch (finish) {
    case "matte": return P({ metalness: 0.08, roughness: 0.72, envMapIntensity: 0.7 });
    case "satin": return P({ metalness: 0.28, roughness: 0.46, clearcoat: 0.4, clearcoatRoughness: 0.32 });
    case "metallic": return P({ metalness: 0.78, roughness: 0.32, clearcoat: 1, clearcoatRoughness: 0.04 });
    case "pearl": return P({ metalness: 0.45, roughness: 0.24, clearcoat: 1, clearcoatRoughness: 0.03, iridescence: 0.55, iridescenceIOR: 1.6, iridescenceThicknessRange: [180, 520] });
    case "chrome": return P({ metalness: 1, roughness: 0.06, clearcoat: 0.6, clearcoatRoughness: 0.02, envMapIntensity: 1.6 });
    default: return P({ metalness: 0.32, roughness: 0.3, clearcoat: 1, clearcoatRoughness: 0.03 });
  }
}

    camera.position.lerpVectors(flight.from, flight.to, e);
    if (f >= 1) flight = null;
  }
  controls.update();
  renderer.render(scene, camera);

const look = () => looks.get(currentKey) ?? factoryLook();
const store = { get(k) { try { return localStorage.getItem(k); } catch { return null; } }, set(k, v) { try { localStorage.setItem(k, v); } catch { /* storage unavailable */ } } };

function buildGarage() {
  $("garage-count").textContent = `Garage · ${GARAGE.length}`;
  const ul = $("garage");
  for (const e of GARAGE) {
    const li = document.createElement("li");
    const b = document.createElement("button");
    b.type = "button"; b.dataset.key = e.key;
    b.innerHTML = `<span class="n"></span><span class="m"></span>`;
    b.querySelector(".n").textContent = e.label;
    b.querySelector(".m").textContent = e.detail;
    b.addEventListener("click", () => selectCar(e.key));
    li.append(b); ul.append(li);
  }
}

function load(entry) {
  if (cache.has(entry.key)) return Promise.resolve(cache.get(entry.key));
  $("loading").hidden = false;
  return new Promise((resolve, reject) => {
    loader.load(entry.file, (gltf) => { const p = prepare(gltf, entry); cache.set(entry.key, p); resolve(p); }, (ev) => {
      const share = ev.total ? ev.loaded / ev.total : 0;
      $("load-pct").textContent = `${Math.round(share * 100)}%${ev.total ? ` of ${(ev.total / 1e6).toFixed(1)} MB` : ""}`;
      $("load-fill").style.width = `${share * 100}%`;
    }, reject);
  }).finally(() => { $("loading").hidden = true; });
}

let requested = null;
async function selectCar(k) {
  requested = k;
  const entry = GARAGE.find((e) => e.key === k) ?? GARAGE[0];
  document.querySelectorAll("#garage button").forEach((b) => b.setAttribute("aria-current", String(b.dataset.key === entry.key)));
  $("car-title").textContent = entry.label;
  $("car-sub").textContent = entry.detail;
  let prepared;
  try { prepared = await load(entry); } catch (err) {
    $("car-sub").textContent = "This model could not be loaded. Pick it again to retry.";
    console.warn(err); return;
  }
  if (requested !== k) return;
  if (current) scene.remove(current.car);
  current = prepared; currentKey = entry.key;
  scene.add(current.car);
  blob.scale.set(current.size.x * 1.25, current.size.z * 1.12, 1);
  store.set("carbon-garage-car", entry.key);
  credit(entry);
  applyLook();
  renderPanel();
  flyTo(view, !flight && !camera.position.lengthSq());
}

function credit(e) {
  const el = $("credit");
  el.textContent = "";
  const a = (href, text) => { const x = document.createElement("a"); x.href = href; x.target = "_blank"; x.rel = "noopener"; x.textContent = text; return x; };
  if (e.own) {
    el.append("3D model built in code from Porsche's published dimensions and studio photographs (", a(e.source, "build script"), ").");
    return;
  }
  el.append("3D model: ", a(e.source, e.name), " by ", e.authorUrl ? a(e.authorUrl, e.author) : e.author, ", ", a(e.licenseUrl, e.license), ".");
}

let paintMat = null, tintMats = [], lampMats = [];
function applyLook() {
  const L = look();
  paintMat?.dispose(); paintMat = null;
  if (L.paint && current.paintable) { paintMat = paintMaterial(L.paint.hex, L.paint.finish); setSlots(current.paintSlots, paintMat); }
  else setSlots(current.paintSlots, null);

  tintMats.forEach((m) => m.dispose()); tintMats = [];
  if (L.tint) {
    const [opacity, color] = TINTS[L.tint];
    for (const s of current.glassSlots) {
      const m = s.original.clone(); Object.assign(m, { transparent: true, opacity, depthWrite: false }); m.color?.set(color);
      tintMats.push(m); setSlots([s], m);
    }
  } else setSlots(current.glassSlots, null);

  lampMats.forEach((m) => m.dispose()); lampMats = [];
  if (L.lights) {
    const made = new Map();
    for (const s of current.lampSlots) {
      const k = `${s.original.uuid}:${s.front}`;
      let m = made.get(k);
      if (!m) { m = s.original.clone(); m.emissive = new THREE.Color(s.front ? "#f2f6ff" : "#ff1a12"); m.emissiveIntensity = s.front ? 2.6 : 2.2; made.set(k, m); lampMats.push(m); }
      setSlots([s], m);
    }
  } else setSlots(current.lampSlots, null);

  current.body.position.y = current.hasWheels ? L.ride / 1000 : 0;
}

function setLook(patch) {
  looks.set(currentKey, { ...look(), ...patch });
  applyLook();
  renderPanel();
}

function chip(label, pressed, onClick, disabled = false) {
  const b = document.createElement("button");
  b.type = "button"; b.className = "chip"; b.textContent = label;
  b.setAttribute("aria-pressed", String(pressed)); b.disabled = disabled;
  b.addEventListener("click", onClick);
  return b;
}

function renderPanel() {
  const L = look();
  const sw = $("swatches"); sw.textContent = "";
  const factory = document.createElement("button");
  factory.type = "button"; factory.className = "swatch factory"; factory.title = "The model's own paint";
  factory.setAttribute("aria-label", "Factory paint"); factory.setAttribute("aria-pressed", String(!L.paint));
  factory.innerHTML = "<span>Fac</span>";
  factory.addEventListener("click", () => setLook({ paint: null }));
  sw.append(factory);
  for (const [name, hex, finish] of PAINTS) {
    const b = document.createElement("button");
    b.type = "button"; b.className = "swatch"; b.title = `${name} · ${finish}`;
    b.setAttribute("aria-label", `${name}, ${finish}`);
    b.setAttribute("aria-pressed", String(Boolean(L.paint && L.paint.hex === hex && L.paint.finish === finish)));
    b.disabled = !current.paintable;
    const shine = ["metallic", "pearl", "chrome"].includes(finish) ? `radial-gradient(circle at 30% 30%, #ffffff66, ${hex} 45%, ${hex})` : finish === "matte" ? hex : `radial-gradient(circle at 30% 30%, #ffffff33, ${hex} 40%)`;
    b.style.background = shine;
    b.addEventListener("click", () => setLook({ paint: { hex, finish, name } }));
    sw.append(b);
  }
  $("paint-name").textContent = L.paint ? `${L.paint.name ?? "Custom"} · ${L.paint.finish}` : "Factory paint";
  $("paint-note").hidden = current.paintable;
  $("custom-paint").disabled = !current.paintable;
  if (L.paint) $("custom-paint").value = L.paint.hex;

  const fin = $("finishes"); fin.textContent = "";
  for (const f of FINISHES) fin.append(chip(f[0].toUpperCase() + f.slice(1), Boolean(L.paint && L.paint.finish === f), () => setLook({ paint: { hex: L.paint?.hex ?? $("custom-paint").value, finish: f, name: L.paint?.name ?? "Custom" } }), !current.paintable));

  const tints = $("tints"); tints.textContent = "";
  tints.append(chip("Stock", !L.tint, () => setLook({ tint: null })));
  for (const t of Object.keys(TINTS)) tints.append(chip(t[0].toUpperCase() + t.slice(1), L.tint === t, () => setLook({ tint: t }), current.glassSlots.length === 0));

  const lights = $("lights"); lights.textContent = "";
  lights.append(chip("Off", !L.lights, () => setLook({ lights: false })), chip("On", L.lights, () => setLook({ lights: true }), current.lampSlots.length === 0));

  $("ride").value = String(L.ride); $("ride").disabled = !current.hasWheels;
  $("ride-val").textContent = `${L.ride > 0 ? "+" : ""}${L.ride} mm`;
  $("ride-note").hidden = current.hasWheels;
}

$("custom-paint").addEventListener("input", (e) => setLook({ paint: { hex: e.target.value, finish: look().paint?.finish ?? "gloss", name: "Custom" } }));
$("ride").addEventListener("input", (e) => setLook({ ride: Number(e.target.value) }));
$("reset").addEventListener("click", () => setLook(factoryLook()));

function buildViews() {
  const v = $("views");
  for (const [k, text] of VIEWS) {
    const b = document.createElement("button");
    b.type = "button"; b.className = "glassy"; b.textContent = text; b.dataset.view = k;
    b.setAttribute("aria-pressed", String(k === view));
    b.addEventListener("click", () => { view = k; v.querySelectorAll("button").forEach((x) => x.setAttribute("aria-pressed", String(x.dataset.view === k))); flyTo(k); });
    v.append(b);
  }
  const s = $("scenes");
  for (const [k, text] of [["studio", "Studio"], ["dark", "Dark"]]) {
    const b = document.createElement("button");
    b.type = "button"; b.className = "glassy"; b.textContent = text; b.dataset.scene = k;
    b.setAttribute("aria-pressed", String(k === sceneName));
    b.addEventListener("click", () => { sceneName = k; setScene(k); s.querySelectorAll("button").forEach((x) => x.setAttribute("aria-pressed", String(x.dataset.scene === k))); });
    s.append(b);
  }
}

buildGarage();
buildViews();
setScene(sceneName);
resize();
camera.position.set(6, 1.4, 8);
const saved = store.get("carbon-garage-car");
selectCar(GARAGE.some((e) => e.key === saved) ? saved : "ford/mustang");
</script>
